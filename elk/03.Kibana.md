

## 1. Cài đặt
```
cd /home/tuanda
tar -xvzf kibana-8.17.4-linux-x86_64.tar.gz
ln -s /home/tuanda/kibana-8.17.4 /home/tuanda/kibana
timedatectl set-timezone Asia/Ho_Chi_Minh #(mục đích để đồng bộ web kibana search index)
```
File cấu hình 
```
cd /home/tuanda/kibana/config
cat kibana.yml | grep -v '#' | grep -v '^$'
server.port: 5601
server.host: "192.168.88.12"
elasticsearch.hosts: ["http://192.168.88.12:9200"]
elasticsearch.username: "kibana_system"
elasticsearch.password: "123456"
```

Chạy manual
```
# cat start.sh
cat /dev/null  > nohup.out
nohup ./bin/kibana &

# cat stop.sh
ps aux | grep node | grep -v grep  | awk '{print $2}' | xargs -i kill -9 {}
```

Systemd (tested)
```
cat << 'EOF' > /etc/systemd/system/kibana.service
[Unit]
Description=Kibana

[Service]
ExecStart= /home/tuanda/kibana/bin/kibana
Restart=always
User=tuanda
Group=tuanda

[Install]
WantedBy=multi-user.target
EOF
```
```
systemctl daemon-reload
systemctl restart kibana
ps aux | grep kibana
```
Nginx
```
cat << 'EOF' > /etc/nginx/conf.d/kibana.conf
server {
    listen 80;

    server_name kibana.tuan.name.vn;

    #auth_basic "Restricted Access";
    #auth_basic_user_file /etc/nginx/htpasswd.users;

    location / {
        proxy_pass http://192.168.88.18:5601/;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        proxy_cache_bypass $http_upgrade;        
    }
}
EOF
```
